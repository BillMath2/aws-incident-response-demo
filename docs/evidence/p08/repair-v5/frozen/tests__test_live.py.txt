def test_native_response_bounds_execution_and_rejects_unknown_or_invalid_calls():
    request = SimpleNamespace(phase="decide", variant="V1", evidence=())
    tools = response_tools(request)
    valid = {"name": "get_service_health", "args": {"service_id": "checkout-api"}}

    def reply(calls, reason="tool_use"):
        return SimpleNamespace(tool_calls=calls, response_metadata={"stopReason": reason})

    assert json.loads(decode_response(reply([valid]), request, tools))["kind"] == "tool"
    assert decode_response(reply([valid, valid]), request, tools) == decode_response(
        reply([valid]), request, tools
    )
    for invalid in (
        reply([valid] * 5),
        reply([valid], "end_turn"),
        reply([{"name": "executor", "args": {}}]),
        reply([{"name": "get_service_health", "args": {"service_id": "other"}}]),
        reply(
            [{"name": "get_service_health", "args": {"service_id": "checkout-api", "extra": True}}]
        ),
    ):
        with pytest.raises(ValueError):
            decode_response(invalid, request, tools)


def test_live_schema_exposes_only_the_current_graph_phase():
    def names_for(phase, variant="V1"):
        return {
            t["function"]["name"]
            for t in response_tools(SimpleNamespace(phase=phase, variant=variant, evidence=()))
        }

    assert names_for("decide") == {
        "get_service_health",
        "get_recent_changes",
        "get_recent_logs",
        "finish_investigation",
        "retrieve_runbook",
    }
    assert names_for("final") == {"finish_investigation"}
    assert names_for("seed", "V2") == {"submit_search"}
    bounded = response_tools(SimpleNamespace(phase="update", variant="V2", evidence=(), round=2))
    assert bounded[0]["function"]["parameters"]["properties"]["next_check"] == {"type": "null"}
    names = ("get_service_health", "get_recent_changes", "get_recent_logs", "retrieve_runbook")
    evidence = tuple(SimpleNamespace(source=name) for name in names)
    assert response_phase(SimpleNamespace(phase="decide", evidence=evidence[:2])) == "decide"
    assert response_phase(SimpleNamespace(phase="decide", evidence=evidence)) == "final"
    assert response_phase(SimpleNamespace(phase="update", evidence=evidence)) == "update"

    from incident_demo.investigator.providers import FixtureTools

        e
        for name in (

            "get_recent_changes",

    }
    reply = SimpleNamespace(
        tool_calls=[{"name": "finish_investigation", "args": args}],
        response_metadata={"stopReason": "tool_use"},
    )
    assert (
        json.loads(decode_response(reply, req, response_tools(req)))["selected_candidate_id"]
        is None
    )
    args["selected_candidate_id"] = ""

@pytest.mark.parametrize(
    "model,cost", [("us.amazon.nova-lite-v1:0", 75), ("us.amazon.nova-pro-v1:0", 250)]
)
def test_approved_cloud_ceiling_retains_atomic_usage_and_batch_limit(model, cost):
    from incident_demo.live.budget import GLOBAL_CEILING_CENTS, CloudBudget

    calls = []
    client = SimpleNamespace(
        transact_write_items=lambda **kw: calls.append(kw),
        update_item=lambda **kw: calls.append(kw),
    )
    sdk = SimpleNamespace(
        resource=lambda *a, **kw: SimpleNamespace(meta=SimpleNamespace(client=client))
    )
    budget = CloudBudget(sdk, "budget")
    req = request().model_copy(update={"settings": LiveSettings(model=model)})
    budget.reserve(req)
    transaction = calls[0]["TransactItems"]
    global_update = transaction[0]["Update"]
    values = global_update["ExpressionAttributeValues"]
    # Exact boundary: after this reservation, the protected $10 and existing 10 cents remain.
    assert values[":ceiling"] + cost == GLOBAL_CEILING_CENTS == 25000 - 1000 - 10
    assert values[":cost"] == cost
    assert "ADD reserved_cents :cost" in global_update["UpdateExpression"]
    assert "lease_until < :now" in global_update["ConditionExpression"]
    assert transaction[1]["Update"]["ExpressionAttributeValues"][":ceiling"] + cost == 1000
    assert transaction[2]["Put"]["ConditionExpression"] == "attribute_not_exists(pk)"
    budget.release_lease(req)
    assert calls[-1]["UpdateExpression"] == "SET lease_until = :expired"


def test_search_native_tag_and_checks_preserve_bounds():
    req = SimpleNamespace(
        phase="seed",
        variant="V2",
        round=0,
        evidence=tuple(
            SimpleNamespace(source=n) for n in ("get_service_health", "get_recent_changes")
        ),
    )
    args = {
        "candidates": [{"candidate_id": "cause-a", "cause": "Deployment regression"}],
        "next_check": {"name": "get_recent_logs", "arguments": {"service_id": "checkout-api"}},
        "check_purpose": "Inspect corroborating logs",
    }

    def decode():
        reply = SimpleNamespace(
            response_metadata={"stopReason": "tool_use"},
            tool_calls=[{"name": "submit_search", "args": args}],
        )
        return json.loads(decode_response(reply, req, response_tools(req)))

    parameters = response_tools(req)[0]["function"]["parameters"]
    assert "kind" not in parameters["properties"]
    checks = parameters["properties"]["next_check"]["anyOf"][0]["oneOf"]
    assert {s["properties"]["name"]["const"] for s in checks} == {
        "get_recent_logs",
        "retrieve_runbook",
    }
    assert decode()["kind"] == "search"
    args["kind"] = "finish"
    with pytest.raises(ValueError, match="tag mismatch"):
        decode()
    args.pop("kind")
    args["next_check"]["name"] = "get_service_health"
    with pytest.raises(ValueError, match="already collected"):
        decode()
    args["next_check"]["name"] = "get_recent_logs"
    req.round = 2
    with pytest.raises(ValueError, match="exhausted"):
        decode()
    args["next_check"] = None
    assert decode()["next_check"] is None
    args["check_purpose"] = ""
    with pytest.raises(ValidationError):
        decode()

def test_native_search_completes_two_distinct_checks_before_final(root):
    from incident_demo.corpus import load_fixture, load_knowledge
    from incident_demo.investigator.engine import Engine
    from incident_demo.investigator.providers import FixtureTools, OfflineProvider

    phases = []

    class NativeScript:
        def respond(self, req, *, timeout_seconds):
            phases.append((req.phase, req.round))
            decision = json.loads(OfflineProvider().respond(req, timeout_seconds=timeout_seconds))
            if decision["kind"] == "search":
                decision.pop("kind")
                call = {"name": "submit_search", "args": decision}
            else:
                args = decision["investigation"]
                args["selected_candidate_id"] = decision["selected_candidate_id"]
                call = {"name": "finish_investigation", "args": args}
            return decode_response(
                SimpleNamespace(
                    tool_calls=[call],
                    response_metadata={"stopReason": "tool_use"},
                ),
                req,
                response_tools(req),
            )

    fixture = load_fixture(root / "fixtures/cases/case-001.json")
    result = Engine(
        root, "V2", fixture.incident, NativeScript(), FixtureTools(fixture, load_knowledge(root))
    ).run()
    assert result.status == "complete"
    assert result.investigation.outcome == "propose_rollback"
    assert result.provider_calls == result.usage.tool_calls == 4
    assert phases == [("seed", 0), ("update", 1), ("update", 2), ("final", 2)]


@pytest.mark.parametrize(
    "faults,call_limit,retries,expected,calls",
    [
        (["empty_seed"], 6, 1, "finished", 5),
        (["uncited_fact"], 6, 1, "finished", 5),
        (["punctuation"], 6, 1, "finished", 5),
        (["empty_seed", "uncited_fact"], 6, 1, "model_output_invalid", 5),
        (["punctuation"], 4, 1, "model_budget_exhausted", 4),
        (["empty_seed"], 6, 0, "model_output_invalid", 1),
    ],
)
def test_native_format_correction_is_counted_bounded_and_never_echoes_content(
    root, faults, call_limit, retries, expected, calls
):
    from incident_demo.contracts.experiments import Limits, Settings
    from incident_demo.corpus import load_fixture, load_knowledge
    from incident_demo.investigator.engine import Engine
    from incident_demo.investigator.providers import FixtureTools, OfflineProvider

    pending = list(faults)
    requests = []

    class Native:
        def respond(self, req, *, timeout_seconds):
            requests.append(req)
            decision = json.loads(OfflineProvider().respond(req, timeout_seconds=timeout_seconds))
            if decision["kind"] == "search":
                decision.pop("kind")
                call = {"name": "submit_search", "args": decision}
                if pending and pending[0] == "empty_seed" and req.phase == "seed":
                    pending.pop(0)
                    decision["candidates"] = []
                    decision["PRIVATE-KEY"] = "DEMO_CANARY_UNTRUSTED"
            else:
                args = decision["investigation"]
                args["selected_candidate_id"] = decision["selected_candidate_id"]
                call = {"name": "finish_investigation", "args": args}
                if pending and pending[0] == "uncited_fact":
                    pending.pop(0)
                    args["facts"][0]["evidence_ids"] = []
                elif pending and pending[0] == "punctuation":
                    pending.pop(0)
                    args["next_check"] = "."
            return decode_native_response(
                SimpleNamespace(tool_calls=[call], response_metadata={"stopReason": "tool_use"}),
                req,
                response_tools(req),
            )

    fixture = load_fixture(root / "fixtures/cases/case-001.json")
    result = Engine(
        root,
        "V2",
        fixture.incident,
        Native(),
        FixtureTools(fixture, load_knowledge(root)),
        Settings(limits=Limits(model_calls=call_limit, retries=retries)),
    ).run()
    assert result.stop_reason == expected and result.provider_calls == calls
    assert sum(bool(r.validation_feedback) for r in requests) <= 1
    assert "PRIVATE" not in json.dumps([r.validation_feedback for r in requests])
    assert "DEMO_CANARY_UNTRUSTED" not in result.model_dump_json()
    assert len([t for t in result.trace if t.event == "model_output_repair"]) <= 1
    if expected == "finished":
        assert sum(bool(r.validation_feedback) for r in requests) == 1
        assert result.investigation.outcome == "propose_rollback"
        assert result.investigation.next_check != "."
    if "empty_seed" in faults and len(requests) > 1:
        assert requests[1].phase == "seed" and requests[1].round == 0
        assert requests[1].evidence == requests[0].evidence
        assert requests[2].validation_feedback == ()

    assert stripped["properties"]["next_check"]["type"] == "string"

def test_nova_counting_fallback_is_explicit_and_narrow():
    class Unsupported(Exception):
        pass

    class Client:
        exceptions = SimpleNamespace(ValidationException=Unsupported)

        def count_tokens(self, **kwargs):
            raise Unsupported("The provided model doesn't support counting tokens.")

        def converse(self, **kwargs):
            return {
                "usage": {"inputTokens": 100, "outputTokens": 20},
                "ResponseMetadata": {"RequestId": "model-request"},
            }

    usage = {"model_calls": 0, "input_tokens": 0, "output_tokens": 0}
    events = []
    client = CountedClient(Client(), lambda event, **kwargs: events.append(event), usage)
    client.converse(
        modelId="us.amazon.nova-lite-v1:0", messages=[], inferenceConfig={"maxTokens": 1500}
    )
    assert events == ["token_count_unsupported", "model_started", "model_finished"]
    with pytest.raises(Unsupported):
        client.converse(modelId="other-model", messages=[], inferenceConfig={"maxTokens": 1500})
    with pytest.raises(ValueError, match="byte cap"):
        client.converse(
            modelId="us.amazon.nova-lite-v1:0",
            messages=["x" * 32769],
            inferenceConfig={"maxTokens": 1500},
        )
    assert usage["model_calls"] == 1


def test_unknown_billed_call_is_still_counted():
    class Client:

@pytest.mark.parametrize(
    "response,error",
    [
        ({"status": "error", "retryable": True}, TransientFailure),
        ({"status": "error", "retryable": False}, ToolFailure),
    ],
)
def test_lambda_error_is_not_fabricated_evidence(response, error):
    class Client:
        exceptions = SimpleNamespace(TooManyRequestsException=ConnectionError)

        def invoke(self, **kwargs):
            return {
                "Payload": BytesIO(json.dumps(response).encode()),
                "ResponseMetadata": {"RequestId": "lambda-request"},
            }

    sdk = SimpleNamespace(client=lambda *args, **kwargs: Client())
    tools = LambdaTools(
        sdk, {"get_recent_logs": "allowed-arn"}, request(), lambda *args, **kwargs: None
    )
    with pytest.raises(error):
        tools.invoke(tool_call("get_recent_logs"), timeout_seconds=10)
    with pytest.raises(ToolFailure):
        tools.invoke(tool_call("retrieve_runbook"), timeout_seconds=10)
